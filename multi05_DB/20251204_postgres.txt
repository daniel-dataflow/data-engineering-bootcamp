-- 2019년 11월 12월에 사용한 내역들 전체 출력
select * from card where usedate ='2019-11-12';

-- 2019년도에 사용한 전체내역들 출력
select * from card where usedate >= '2019/01/01' and usedate <= '2019/12/31';
select * from card where usedate between '2019/01/01' and '2019/12/31';
-- select * from card where year(usedate) =2019; X
select * from card where usedate::varchar like '2019%';
SELECT * FROM card WHERE TO_CHAR(usedate,'yyyy') = '2019';
SELECT * FROM card WHERE EXTRACT(YEAR FROM usedate) = 2019;
