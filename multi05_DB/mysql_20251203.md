-- inline view : from 뒤에 사용 = join 뒤에 table처럼 작동

Select dept_name, emp_no salary 

from employees join dept_emp using(emp_no)

Join departments using(dept_no)

Join salaries using(emp_no)

Join (select dept_no, max(salary) as maxsal From dept_emp

Join salaries using(emp_no) group by dept_no) as maxsalary using(dept_no)

Where salary = maxsalary.maxsal;
