// filtering, grouping, aggre - 걸러내고 그룹화 시켜서 집합으로 나타내준다.
// filtering
db.score.aggregate({$match:{"kor":{$gt:80}}});

// grouping
// select * from score group by test;
db.score.aggregate({$group:{"_id":"$test"}});

// aggre
// select test, avg(kor) from score group by test;
db.score.aggregate({$group:{"_id":"$test", "average": {$avg:"$kor"}}});
// select avg(kor) from score;
db.score.aggregate({$group:{"_id":null, "average": {$avg:"$kor"}}});

// sum
db.score.aggregate({$group:{"_id":null, "sum": {$sum:"$kor"}}});
db.score.aggregate({$group:{"_id":"$test", "sum": {$sum:"$kor"}}});

// null - min, max
db.score.aggregate({$group:{"_id":null, "sum": {$min:"$kor"}}});
db.score.aggregate({$group:{"_id":null, "sum": {$max:"$kor"}}});

// $test - min, max
db.score.aggregate({$group:{"_id":"$test", "sum": {$min:"$kor"}}});
db.score.aggregate({$group:{"_id":"$test", "sum": {$max:"$kor"}}});

// 배열 형태에 스테이지가 들어가고 그안에 도큐먼트형태로 들어가고 
// 필터형태로 보여주겠다.
db.score.aggregate({$project:{"_id":0, "name":1, "kor":1}});


// 파이프라인 - 스테이지의 결과가 다음 스테이지의 입력으로 들어간다.
db.score.aggregate(
    [
        {$match: {"test": "final"}},
        {$project: {"name":1, "eng":1}},
        {$group:{"_id":"final exam", "eng avg": {$avg: "$eng"}}}
    ]
);





// 점수(kor, eng, math)의 합계가 250점 이상인 사람들의 이름과 점수 합계를 내림차순으로
// $project: 다음에 무슨값을 보내줄지 결정한다.
// "$kor" : 각 필드의 값 하나하나를 사용해야 할때 $를 붙인다.
// $match: 맞는 조건을 넣어준다,total이 250이상인 값을 다음으로 넘긴다.
// $sort: 값을 정렬해서 보여줘라 (-1 : 큰수부터)

db.score.aggregate([
    {$project: {"name":1, "total": {$add: ["$kor", "$eng", "$math"]}}},
    {$match:{"total":{$gte:250}}},
    {$sort:{"total":-1}}
]);

// "학생 각각의 평균" 에 대한 전체 평균을 구하고, 반올림
// $round:["$average", 0] : 0: 보여지는 자릿수
db.score.aggregate([
    {$group:{"_id":null, "average":{$avg:{$avg:["$kor","$eng","$math"]}}}},
    {$project:{"_id":0,"average":{$round:["$average", 0]}}}
]);


// Map-Reduce : 묶어서 줄어준다. 
// Data Modeling


// explain : 실행계획
// winningPlan : 최종적으로 선택한 것
// rejectedPlans: 선택하지 않은 것
db.score.find({"name":"이순신"}).explain();

// executionStats : 실제 실행결과
// totalKeysExamined : 인덱스를 사용해서 찾을 때 사용한 인덱스의 갯수
db.score.find({"name":"이순신"}).explain("executionStats");

// allPlansExecution : winningPlan 와 rejectedPlans 마저도 모두 실행한 결과를 알려준다.
db.score.find({"name":"이순신"}).explain("allPlansExecution");
