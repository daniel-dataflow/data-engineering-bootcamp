-- rollup, cube, grouping sets

-- rollup (a,b)
-- a, b 집계
-- a, null 집계
-- null, null 집계


select main_cate, sub_cate, sum(amount)
from  card
group by rollup(main_cate, sub_cate)
order by main_cate, sub_cate;


-- cube (a, b)
-- a, b 집계
-- a, null 집계
-- null, b 집계
-- null, null 집계

select main_cate, sub_cate, sum(amount)
from  card
group by cube(main_cate, sub_cate)
order by main_cate, sub_cate;



-- grouping sets (컬럼들) : 컬럼들에 대한 집계

from card

-- inline view : 서브 쿼리가 테이블처럼 사용되었다. (가상 테이블)
-- 스토어가 널이 아니고 스토어별로 그룹지어서 
-- 스토어 별로 금액의 총합계를 구하고 스토어의 방문수를 구하는데 컬럼에 별칭을 줘서 값을 나오게 하였다.
select store, sum_amount, cnt
from
    (select store, sum(amount) as sum_amount, count(*) as cnt
    from card
    where store is not null
    group by store)
order by 3 desc;



-- self join
-- 같은 달, 같은 금액을 쓴 스토어를 보고 싶다.
select c1.store, c1.usedate, c2.store, c2.usedate, c1.amount
from card c1 join card c2 on c1.amount = c2.amount
where c1.id < c2.id
and to_char(c1.usedate, 'yyyy-mm') = '2019-03'
and to_char(c2.usedate, 'yyyy-mm') = '2019-03'
and c1.store is not null
and c2.store is not null
order by 2, 4;



-- window function
-- rank : 같은 등수 이후 중복된 만큼 건너 띄고 다음 순번
-- dense_rank : 같은  등수 이후 다음 순번
-- row_number : 무조건 순번대로
select main_cate, sub_cate,
    rank() over (order by amount desc),
    dense_rank() over (order by amount desc),
    row_number() over (order by amount desc),
    amount
from card
where main_cate = '문화'
limit 10;

-- 치킨 카테고리에 있는 것 나오면서 많이 사용한 금액순으로 랭크
select store, amount, rank() over (partition by store order by amount desc)
from card
where sub_cate='치킨';


-- 메인카테별로 어떤 서부카테가 가장 돈을 많이 사용했는지 1,2,3위만 랭크

with cate_rank as (
    select main_cate, sub_cate,
    rank() over (partition by main_cate order by amount desc) as rnk,
    dense_rank() over (partition by main_cate order by amount desc) as dnsmk,
    row_number() over (partition by main_cate order by amount desc) as rownum,
    amount
    from card
)
select main_cate, sub_cate, amount, rnk, dnsmk, rownum
from cate_rank
where rnk <=3
order by main_cate, amount desc;

create or replace function avg_year(
    target_year int
)
returns decimal                         -- return type (없으면 void)
language plpgsql                        
as $$                                   -- delimiter 변경
declare                                 --  함수 내에서 사용할 변수 선언
    avg_amount decimal :=0.0;           --  변수 타입 [:=값]
begin
    select round(avg(amount), 2) into avg_amount -- 반올림 소수점 2째자리
    from card
    where extract (year from usedate) = target_year;

    return avg_amount;
end;
$$;
