-- window function
-- 함수 () over ([partition by 컬럼] [order by 컬럼] [rows/range])

Select id, grade, score, 
    rank() over(order by score desc) as rnk,
    Dense_rank() over (order by score desc) as dnsrnk,
    Row_number() over (order by score desc) as rownum
from student;

-- rnk : 이미 메겨진 순서 이후의 다음 순서가 나타난다,
-- dnsrnk :  같은 등수일 경우 그다음 순번을 메겨준다.
-- ronum :  같은 등수일 경우 무조건 넘버링을 한다. 아이디를 기준으로 나눈다.
-- order by : 큰 스코어 기준이 될거야. 



Select id, grade, score, 
    rank() over (partition by grade order by score desc) as rnk,
    Dense_rank() over (partition by grade order by score desc) as dnsernk,
    Row_number() over (partition by grade order by score desc) as rownum
from student;

-- rnk : 이미 메겨진 순서 이후의 다음 순서가 나타난다,
-- dnsrnk :  같은 등수일 경우 그다음 순번을 메겨준다.
-- ronum :  같은 등수일 경우 무조건 넘버링을 한다. 아이디를 기준으로 나눈다.
-- partition by : group by grade 묶어준다.  큰 스코어 기준으로 내림차순 될거야. 



-- 집계함수를 윈도우 함수로 사용시!
-- order by 를 사용하면 누적 집계가 된다!

select id, grade, score,
    Sum(score) over (partition by grade order by  id) as cum_sum,
    Sum(score) over(partition by grade) as group_sum,
    Avg(score) over (partition by grade) as group_avg,
    Max(score) over (partition by grade) as group_max,
    Min(score) over (partition by grade) as group_min
from student;

-- cum_sum: 누적으로 합이된다.

-- with table (CTE : Common Table Expression)
-- 가상 테이블에 값을 담아서 하단의 메인 쿼리에서 실행하도록 한다.
with production_dept as(
    select dept_no
    from departments
    where dept_name = 'Production'
),
production_manager as(
    select emp_no
    from dept_manager
    join production_dept as pd using(dept_no)
    where to_date = '9999-01-01'
)
select * 
from employees join production_manager as pm using(emp_no);

-- Mysql 문법에선 group by a, b with rollup;
-- a, b 집계(합계) (a와 b를 같이 묶어서 집계)
-- a, null 집계 (a에 대한 집계)

from employees join titles using (emp_no)
where to_date = '9999-01-01'

--parameter
-- in : 프로시저에 입력되는 값을 받을 것이다.
-- target : 변수 명
-- date : 데이터 유형
delimiter $$
create procedure d_day(in target date)
begin
    select datediff(target, now());
end$$
delimiter ;

call d_day('2026-03-13');


-- out  : return
drop procedure d_day;

delimiter $$
create procedure d_day(out target int)
begin 
    select datediff('2026-03-13', now()) into target;
end$$
delimiter ;

call d_day(@result);
select @result;

-- inout :  입력 받은 값 -> 처리후 -> 다시 리턴
delimiter $$
create procedure mycounter(inout i int)
begin
    set i = 1 + 2;
end$$
delimiter ;

set @i = 1;
call mycounter(@i);
select @i;
