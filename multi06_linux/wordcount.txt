import sys, re
from pyspark import SparkContext

# RDD 관리 - 실행시켜주는 역할
sc = SparkContext() 

# 3개라면 실행시켜라
if len(sys.argv) != 3: 
    print('wordcount.py input_file output_dir 형태로 실행해 주세요')
    sys.exit(0)
else:
    input_file = sys.argv[1]
    output_dir = sys.argv[2]

wordcount = sc.textFile(input_file)\                            # textFile : 들어온 파일을 RDD로 생성하라
              .repartition(10)\                                 # repartition : 파티션개수를 뒤에 지정된 숫자로 재분배
              .filter(lambda x: len(x) > 0)\                    # filter : 길이가 0보다 큰것만 뽑아내자(빈 문자열 제거)
              .flatMap(lambda x: re.split(r'\W+', x))\          # flatMap : 여러개의 출력물을 모두 펼쳐서 보여준다
                                                                # re.split : 정규식을 통해 문자를 ,로 구별해서 보여준다.
              .filter(lambda x: len(x) > 0)\                    # filter: 다시한번 빈문자열 제거
              .map(lambda x: (x.lower(), 1))\                   # map : 모두 소문자로 변환하고 key, value 형태인 튜플로 변환한다. 
              .reduceByKey(lambda x, y: x + y)\                 # reduceByKey : 같은 키를 가진 것의 갯수를 더해서 보여주자.
              .map(lambda x: (x[1], x[0]))\                     # map : key와 value의 순서를 서로 뒤집어 바꾼다.
              .sortByKey(ascending=False)\                      # sortByKey : 내림차순으로 보여주자
              .persist()                                        # persist : 지금까지 한 작업을 메모리에 저장한다.(action에서 재계산 방지 / 여러번 재 사용할 경우 성능 최적화에 필수)

wordcount.saveAsTextFile(output_dir)            # DAG 생성-> Shuffle기준 Stage분리 ->Task 분해-> Executor에 분산 실행 -> 파티션 수만큼 파일 생성

top10 = wordcount.take(10)                      # 필요한 파티션만 스캔하여 상위 10개만 수집 / 결과는 Drive 메모리에 반환
result = []                                     # 결과 리스트를 만든다.
for counts in top10:                            # 상위 10개를 반복해서 수행한다.
    result.append(counts[1])                    # 튜플의 2번째 요소를 결과 리스트에 넣는다.
print(result)                                   # 결과 출력
