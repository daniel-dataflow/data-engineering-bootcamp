-- Production 부서의 현재 매니저를 찾아서, 해당 사원의 모든 정보를 출력
-- departments (dept_no, dept_name) -> dept_manager (dept_no, emp_no) -> employees
select *
from employees
where emp_no = 
	(select emp_no
	from dept_manager
	where dept_no = 
		(select dept_no
		from departments
		where dept_name = 'Production')
	and to_date = '9999-01-01');

-- 현재 사원들을 대상으로, 평균 월급보다 더 많은 월급을 받는 사원의 수 출력

-- inline view : from 뒤에 사용 = table처럼 작동

select dept_name, emp_no, salary 
from employees join dept_emp using(emp_no)
	join departments using(dept_no)
	join salaries using(emp_no)
	join (select dept_no, max(salary) as maxsal
		from dept_emp
		join salaries using(emp_no)
		group by dept_no) as maxsalary using(dept_no)
where salary = maxsalary.maxsal;

-- self join
-- 같은 월급을 받고 있는 사원들 출력 월급 내림차순
select a.emp_no, b.emp_no, a.salary
from salaries as a join salaries as b on a.salary = b.salary
where a.emp_no > b.emp_no 
and a.to_date = '9999-01-01'
and b.to_date = '9999-01-01'
order by 3 desc
limit 1000;
