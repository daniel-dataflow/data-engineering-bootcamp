from pyspark.sql import Row

null_df = sc.parallelize([
	Row(name="You", phone="010-0000-0000", address="Seoul"),
	Row(name="Shin", phone="010-1111-1111", address=None),
	Row(name="Kang", phone=None, address=None)
]).toDF()

null_df.show()

null_df.createOrReplaceTempView("null_table")


from pyspark.sql.functions import coalesce, col

# rdd의 coalesce 함수는 repartition에 사용된다!!
# dataframe의 coalesce 함수는 null이 아닌 첫 번째 컬럼값

null_df.select(coalesce(col("address"), col("phone"))).show()

spark.sql("""
	select ifnull(null, 'value'), nullif('same', 'same'),
	nvl('notnull', 'value'), nvl2(null, 'value1', 'value2')
""").show()


spark.sql("""
	select ifnull(null, 'value'), nullif('same', 'same'),
	nvl('notnull', 'value'), nvl(null, 'value'), 
	nvl2('notnull', 'value1', 'value2'), nvl2(null, 'value1', 'value2')
""").show()

# na : DataFrameNaFunction 리턴
# DataFrameNaFunction : drop, fill, replace

null_df.na.drop().show()

# any : 컬럼의 값중 하나라도 null이면 row 제거
# all : 컬럼의 모든 값이 null이면 row 제거
null_df.na.drop("all", subset=["phone"]).show()
null_df.na.drop("all", subset=["address"]).show()
null_df.na.drop("all", subset=["phone", "address"]).show()
null_df.na.drop("any", subset=["phone", "address"]).show()

# fill
null_df.na.fill("n/a").show()

null_df.na.fill('n/a', subset=["phone"]).show()
null_df.na.fill('n/a', subset=["address"]).show()

null_df.na.fill('n/a', subset=["phone", "address"]).show()

default_value = {"phone": "070-7777-7777", "address": "street"}
null_df.na.fill(default_value).show()

# replace
null_df.na.replace(["Seoul"], ["서울"], "address").show()

card = spark.read.format("csv").option("header", "true")
		.option("inferSchema", "true").load("./data/card_data/csv/202010.csv")

card.createOrReplaceTempView("card")

card.show()
spark.sql(" select * from card ").show()


# 구조체 : dataframe 내부의 dataframe

card.selectExpr("(`이용일`, `금액`) as complex", "*").show()

card.selectExpr("struct(`이용일`, `금액`) as complex", "*").printSchema()

from pyspark.sql.functions import struct

complex_df = card.select(struct("이용일", "금액").alias("complex"))
complex_df.show()

complex_df.createOrReplaceTempView("complex_table")
spark.sql(" select * from complex_table ").show()


complex_df.select("complex.이용일").show()
complex_df.select(col("complex").getField("이용일")).show()

spark.sql("""
	select complex.`이용일`
	from complex_table
""").show()

complex_df.select("complex.*").show()

from pyspark.sql.functions import split

card.select(col("중분류"), split(col("중분류"), " ")).where(col("대분류") == "교통").show()

spark.sql("""
	select split(`중분류`, " ")
	from card where `대분류` = "교통"
""").show()

card.select(split(col("중분류"), " ").alias("arraycol"))
	.selectExpr("arraycol[0]").where(col("대분류") == "교통").show()

spark.sql("""
	select split(`중분류`, " ")[0] as arraycol
	from card where `대분류` = "교통"
""").show()

from pyspark.sql.functions import array_contains

card.select(array_contains(split(col("중분류"), " "), "버스")).where(col("대분류") == "교통").show()

spark.sql("""
	select array_contains(split(`중분류`, " "), "버스")
	from card where `대분류` = "교통"
""").show()

# explode : 1정규화
from pyspark.sql.functions import explode

card.withColumn("splitted", split(col("중분류"), " "))
	.select("중분류", "splitted").where(col("대분류") == "교통").show()

card.withColumn("splitted", split(col("중분류"), " "))
	.withColumn("exploded", explode(col("splitted")))
	.select("중분류", "splitted", "exploded").where(col("대분류") == "교통").show()


# map {key -> value}

from pyspark.sql.functions import create_map

card.select(create_map(col("이용일"), col("금액")).alias("complexmap")).show()

spark.sql("""
	select map(`이용일`, `금액`)
	from card
""").show()

# 키를 통해 밸류에 접근해보자!!

card.select(create_map(col("이용일"), col("금액")).alias("complexmap"))
	.selectExpr("complexmap['20201002']").show()


spark.sql("""
	select map(`이용일`, `금액`)["20201002"]
	from card
""").show()

# explode : map 객체를 key / value 분리
card.select(create_map(col("이용일"), col("금액")).alias("complexmap"))
	.selectExpr("explode(complexmap)").show()

spark.sql("""
	select explode(map(`이용일`, `금액`))
	from card
""").show()

exit()
