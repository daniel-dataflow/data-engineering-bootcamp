# maingcategory가 '식사' 인 데이터 검색
GET card/_search
{
	"query": {
		"term": {
			"maincategory": {
				"value": "식사"
			}
		}
	}
}

# 위와 동일
GET card/_search
{
	"query": {
		"bool": {
			"must": [
				{"match": {
					"maincategory" : "식사"
					}
				}
			]
		}
	}
}

# 또!
GET card/_search
{
	"query": {
		"bool": {
			"filter": [
				{"term": {
					"maincategory" : "식사"
				}}
			]
		}
	}
}

# match + filter
GET card/_search
{
	"query": {
		"bool": {
			"must": [
				{"match": 
					{"maincategory": "식사"}}
			],
			"filter": [
				{"term": {
					"subcategory": "닭강정"
				}}
			]
		}
	}
}

# postdate가 202011201 인 데이터들 중에 amount가 5000 이상인 데이터들 검색
GET card/_search
{
	"query": {
		"bool": {
			"must": [{"match": {
						"postdate": "20211201"
					}
				}
			],
			"filter": [{"range": {
						"amount": {
							"gte": 5000
						}
					}
				}
			]
		}
	}
}

# must : and
# should : or
GET card/_search
{
	"query": {
		"bool": {
			"should": [
				{"match":
					{"postdate": "20211204"}
				},
				{"match":
					{"postdate": "20211207"}
				}
			]
		}
	}
}

# 위와 동일
GET card/_search
{
	"query": {
		"match": {
			"postdate": {
				"query": "20211204 20211207",
				"operator": "or"
			}
		}
	}
}
