-- window function
-- 함수() over ([partition by 컬럼] [order by 컬럼] [rows/range])
select id, grade, score,
	rank() over (order by score desc) as rnk,
	dense_rank() over (order by score desc) as dnsrnk,
	row_number() over (order by score desc) as rownum
from student;

select id, grade, score,
	rank() over (partition by grade order by score desc) as rnk,
	dense_rank() over (partition by grade order by score desc) as dnsernk,
	row_number() over (partition by grade order by score desc) as rownum
from student; 

-- 집계함수를 윈도우함수로 사용 시!
-- order by를 사용하면 누적 집계가 된다!
select id, grade, score,
	sum(score) over (partition by grade order by id) as cum_sum,
	sum(score) over (partition by grade) as group_sum,
	avg(score) over (partition by grade) as group_avg,
	max(score) over (partition by grade) as group_max,
	min(score) over (partition by grade) as group_min
from student;

	select date_format(now(), '%Y년 %m월 %d일 %H시 %i분');
end$$
delimiter ;

call currnow();

-- parameter
delimiter $$
create procedure d_day(in target date)
begin
	select datediff(target, now());
end$$
delimiter ;

call d_day('2026-03-13');

-- return
drop procedure d_day;

delimiter $$
create procedure d_day(out target int)
begin
	select datediff('2026-03-13', now()) into target;
end$$
delimiter ;

call d_day(@result);
select @result;

end$$

set @i = 1;
