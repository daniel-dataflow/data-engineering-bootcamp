// get
curl https://jsonplaceholder.typicode.com/posts
// get (1개만)
curl https://jsonplaceholder.typicode.com/posts/4

// -X (method 명시)
curl -X GET https://jsonplaceholder.typicode.com/posts/4

// -i (header + body)
curl -i https://jsonplaceholder.typicode.com/posts/4

// post
// -d : data 
curl -d "title=test&body=insert" https://jsonplaceholder.typicode.com/posts

// -H : header 변경
// json data format : "{\"key\": \"val\"}"

curl -H "Content-Type: application/json" -d "{\"title\": \"test\", \"body\": \"abcd\" }" https://jsonplaceholder.typicode.com/posts

// patch
curl -X PATCH -d "title=test" https://jsonplaceholder.typicode.com/posts/1
// put
curl -X PUT -d "title=test2" https://jsonplaceholder.typicode.com/posts/2

// delete

curl -X DELETE https://jsonplaceholder.typicode.com/posts/3
