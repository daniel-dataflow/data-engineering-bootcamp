from airflow.sdk import DAG, task
from pendulum import yesterday
from airflow.providers.standard.operators.python import PythonOperator
from airflow.providers.standard.operators.bash import BashOperator


@task(task_id="task01")
def hello01():
	print("hello, python!")


@task.bash(task_id="task02")
def hello02():
	return "echo 'hello, bash!' "



with DAG(
	dag_id="air14",
	start_date = yesterday(),
	schedule = None
) as dag:

	hello01() >> hello02()

vim air15.py

from airflow.sdk import dag, task
from pendulum import yesterday
from pprint import pprint


@dag(dag_id="air15", start_date=yesterday(), schedule=None)
def main():

	@task(task_id="task01", do_xcom_push=True)
	def get_context(**kwargs):
		pprint(kwargs)
		return str(kwargs)

	@task.bash(task_id="task02")
	def print_context():
		return """
			echo '{{ task_instance.xcom_pull(task_ids="task01") }}'  >> ~/context.json
			"""

	get_context() >> print_context()

main()

from airflow.sdk import DAG, task
from pendulum import yesterday
from airflow.providers.standard.sensors.filesystem import FileSensor


@task(do_xcom_push=True, multiple_outputs=True)
def python_task():
	crawling = {"key01": "value01", "key02": "value02"}
	return crawling

@task.bash
def bash_task():
	return """
		echo '{{ task_instance.xcom_pull(task_ids="python_task") }}' >> ~/dummy.json
	"""

@task.bash
def result_task():
	return " echo 'success' "


with DAG(
	dag_id="air17",
	start_date=yesterday(),
	schedule=None
)as dag:
	exists_task = FileSensor(task_id="exists_task",
				filepath="/home/big/dummy.json",
				fs_conn_id="fs_default"
				)

	python_task() >> [exists_task, bash_task()] >> result_task()

vim dbconnect.py

from pyspark.sql import SparkSession


spark = SparkSession.builder.master("yarn").appName("airflow using spark").getOrCreate()

# mysql
user = "root"
password = "1234"
url = "jdbc:mysql://localhost:3306/mysql"
driver = "com.mysql.cj.jdbc.Driver"
dbtable = "test"

mysql_df = spark.read.format("jdbc")
		.options(user=user, password=password, url=url, driver=driver, dbtable=dbtable).load()

# postgre
jdbc_url = "jdbc:postgresql://localhost:5432/big"
properties = {
	"user": "big",
	"password": "",
	"driver": "org.postgresql.Driver"
}

postgre_df = spark.read.jdbc(url=jdbc_url, table="test", properties=properties)


# mongo
mongo_df = spark.read.format("mongo").option("collection", "test").load()
mongo_df = mongo_df.drop("_id")

# redis
redis_df = spark.read.format("org.apache.spark.sql.redis").option("table", "test")
				.option("key.column", "id").option("infer.schema", "true").load()


# union
result_df = mysql_df.union(postgre_df).union(mongo_df).union(redis_df.select("id", "name"))

# hadoop에 저장
result_df.write.format("json").mode("overwrite").save("/result")
