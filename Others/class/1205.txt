-- rollup, cube, grouping sets
-- rollup(a, b)
-- a, b 집계
-- a, null 집계
-- null, null 집계

select main_cate, sub_cate, sum(amount)
from card
group by rollup(main_cate, sub_cate)
order by main_cate, sub_cate;

-- cube(a, b)
-- a, b 집계
-- a, null 집계
-- null, b 집계
-- null, null 집계

select main_cate, sub_cate, sum(amount)
from card
group by cube(main_cate, sub_cate)
order by main_cate, sub_cate;

-- grouping sets (컬럼들) : 컬럼들에 대한 집계
-- grouping sets를 사용해서 rollup 처럼 출력
select main_cate, sub_cate, sum(amount)
from card
group by grouping sets ( (main_cate, sub_cate), (main_cate), ()  )
order by 1, 2;

-- grouping sets 를 사용해서 cube처럼 출력
select main_cate, sub_cate, sum(amount)
from card
group by grouping sets ( (main_cate, sub_cate), (main_cate), (sub_cate), ()  )
order by 1, 2;

select distinct main_cate
from card;

	where store is not null
	group by store)

-- self join

from card c1 join card c2 on c1.amount = c2.amount
where c1.id < c2.id
and to_char(c1.usedate, 'yyyy-mm') = '2019-03'
and to_char(c2.usedate, 'yyyy-mm') = '2019-03'
and c1.store is not null
and c2.store is not null
order by 2, 4;

-- window function

select main_cate, sub_cate,
	rank() over (order by amount desc),
	dense_rank() over (order by amount desc),
	row_number() over (order by amount desc),
	amount
from card
where main_cate = '문화'
limit 10;

select store, amount, rank() over (partition by store order by amount desc)
from card
where sub_cate='치킨';

with cate_rank as (
	select main_cate, sub_cate,
	rank() over (partition by main_cate order by amount desc) as rnk,
	dense_rank() over (partition by main_cate order by amount desc) as dnsrnk,
	row_number() over (partition by main_cate order by amount desc) as rownum,
	amount
	from card
)
select main_cate, sub_cate, amount, rnk, dnsrnk, rownum
from cate_rank
where rnk <= 3
order by main_cate, amount desc;

create or replace function avg_year(
	target_year int
)
returns numeric					-- return type (없으면 void)
language plpgsql
as $$							-- delimiter 변경
declare						-- 함수 내에서 사용할 변수 선언
	avg_amount numeric := 0.0;		-- 변수 타입 [:= 값]
begin
	select round(avg(amount), 2) into avg_amount
	from card
	where extract(year from usedate) = target_year;

	return avg_amount;
end;
$$;
