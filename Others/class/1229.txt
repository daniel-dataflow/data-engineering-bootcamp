from kafka.admin import KafkaAdminClient, NewTopic
from kafka import KafkaProducer


topic_name = "five-topics"
topic_partitions = 5
admin_client = KafkaAdminClient(bootstrap_servers=["localhost:9092"])
new_topic = NewTopic(name=topic_name, num_partitions=topic_partitions, replication_factor=1)

admin_client.create_topics([new_topic])


producer = KafkaProducer(bootstrap_servers=["localhost:9092"])

for i in range(topic_partitions):
	producer.send(topic_name, value=bytes(f"{i}", "utf-8"), partition=i)
	producer.flush()

producer.close()

vim consumer08.py

from kafka import KafkaConsumer
from kafka.coordinator.assignors.roundrobin import RoundRobinPartitionAssignor

from threading import Thread


def myConsumer(name):
	consumer = KafkaConsumer("five-topics",
				group_id="five",
				bootstrap_servers=["localhost:9092"],
				auto_offset_reset="earliest",
				consumer_timeout_ms=10000,
				partition_assignment_strategy=[RoundRobinPartitionAssignor()])

	for msg in consumer:
		print(f"{name} : {msg.value}")

	consumer.close()



if __name__ == "__main__":
	thread01 = Thread(target=myConsumer, args=("consumer01",), name="consumer01")
	thread02 = Thread(target=myConsumer, args=("consumer02",), name="consumer02")
	
	thread01.start()
	thread02.start()
	thread01.join()
	thread02.join()
