wordcount = sc.textFile("./data/shakespeare.txt")
wordcount.take(10)

# 공백 데이터 제거
wordcount = wordcount.filter(lambda x: len(x) > 0)
wordcount.take(10)

# 문자가 아닌 것으로 잘라서 1차원으로
import re

wordcount = wordcount.flatMap(lambda x: re.split(r"\W+", x))
wordcount.take(10)

# 공백 데이터 제거
wordcount = wordcount.filter(lambda x: len(x) > 0)
wordcount.take(10)

# 모양 변경 -> (x.lower(), 1) <- tuple
wordcount = wordcount.map(lambda x: (x.lower(), 1))
wordcount.take(10)

# key를 기준으로 value 연산
wordcount = wordcount.reduceByKey(lambda x, y: x + y)
wordcount.take(10)

# (key, value) -> (value, key) 형태로 변환
wordcount = wordcount.map(lambda x: (x[1], x[0]))
wordcount.take(10)

# key를 기준으로 정렬
wordcount = wordcount.sortByKey(ascending=False)
wordcount.take(10)

from pyspark.sql.functions import max

card.select(max("`금액`")).show()

# expr(표현식)
# col(컬럼명) -> col 함수 사용 시, 한글 컬럼에 ``(backtick) 사용 안해도 됨!
from pyspark.sql.functions import expr, col

card.select(expr("`대분류` as category"), col("금액")).show()
card.select(expr("`대분류`").alias("category"), col("금액")).show()
spark.sql("select `대분류` as category, `금액` from card").show()

# selectExpr = select + expr
card.selectExpr("`대분류` as category", "`금액`").show()

# selectExpr -> 새로운 컬럼 생성도 가능
card.selectExpr("*", "`금액` > 40000").show()
