member = ['CaptainAmerica', 'Thor', 'Hulk', 'IronMan', 'BlackWidow', 'Hawkeye', 'Hulk']
# parallelize : data를 받아서 rdd 생성
avengers = sc.parallelize(member, 2)
avengers.collect()

# distinct
avengers = avengers.distinct()
avengers.collect()

# glom : 파티션 별로 확인
avengers_partition = avengers.glom()
avengers_partition.collect()

# map
avengers_upper = avengers.map(lambda x: x.upper())
avengers_upper.collect()

# flatmap
avengers_flat = avengers.flatMap(lambda x: list(x))
avengers_flat.collect()

avengers_map = avengers.map(lambda x: list(x))
avengers_map.collect()

# sortBy
nums = sc.parallelize([1, 4, 2, 3, 7, 5, 6], 2)
nums.collect()
nums.sortBy(lambda x:x).collect()

# action (rdd -> value)

# collect() : rdd의 모든 데이터 출력

nums.max(), nums.min()

	if len(x) > len(y):
