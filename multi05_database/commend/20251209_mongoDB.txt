// 가까운 순서대로 출력 + 거리까지 출력하자
db.mongobucks.aggregate([
    {
        $geoNear:{
            near:{
                type:"Point",
                coordinates:[127.0736345, 37.5518018]
            },
            distanceField:"distance",
            spherical:true
        }
    }
]);



// "역"이라는 단어가 포함된 좌표들의 거리 (현재 위치 기준)
db.mongobucks.aggregate([
    {
        $geoNear:{
            near:{
                type:"Point",
                coordinates:[127.0736345, 37.5518018]
            },
            spherical:true,
            query:{
                "s_name":/역/
            },
            distanceField:"distance"
        }
    }
]);

                    ]
                ]
            }
        }
    }
});


// 현재 위치 기준으로 0.5 mile 반경의 원을 그려서 안에 있는 데이터 출력
// 지구의 평균 반경을 mile 단위로 나타낸 상수: 3963.2
// 1 mile = 약 1.6km
db.mongobucks.find({
    "location":{
        $geoWithin:{
            $centerSphere:[
                [127.0736345, 37.5518018],
                0.5 / 3963.2

        }
    }
