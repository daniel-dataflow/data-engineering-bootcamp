# streams type이 만들어졌다. (불편해서) 저장되어 있다. 
# 메세지가 구독자가 없어도 날라가기지 않는다.
# * :  아이디 (random 하게 만들어라)
# msg : key
# "a" : value
xadd channel1 * msg "a"
xadd channel1 * msg "b"
xadd channel1 * msg "c"

# 새창에서 
# 채널 1에서 작은아이디부터 가장 큰 아이디까지 모든 메세지 받아줘
xrange channel1 - +

# xread : 읽어온다 
# 0 : 무한정 대기, 
#현재 스트림(channel1) 아이디 다음꺼부터 읽어오겠다.
xread block 0 streams channel1 $



# 원래창에서 
xadd channel1 * msg "배달 시작했습니다."

# consumer group 
# 한명만 받아도 모두 읽은거로 친다. 중복 방지를 위해
# sgroup : 구독하고 있는 그룹
xgroup create channel1 sgroup $
xadd channel1 * msg  "배달 완료 했습니다."

# 새창에서
# 그룹으로 읽을건데 consumer1라는애가 하나만 읽어올건데 읽은거 다음의 메세지만 가져올께 
xreadgroup group sgroup consumer1 count 1 streams channel1 >



# 원래창에서
xadd channel1 * msg "키보드 1번"
xadd channel1 * msg "키보드 2번"
xadd channel1 * msg "키보드 3번"

# 새창에서
xreadgroup group sgroup consumer1 count 2 streams channel1 >


# 원래창에서
# count 1 인경우는 생략 가능
xreadgroup group sgroup consumer2 count 1 streams channel1 >
